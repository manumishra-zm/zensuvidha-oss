# This is what the DEFAULT config.yaml expects. Both packages pull torch (~530MB),

# runs: the OS voice speaks instead, and the speaker gate fails open.
kokoro>=0.9                # Apache-2.0 TTS — the default tts.provider. In-process, so it
                           # scales with the text (385ms) where macOS `say` costs ~1064ms
                           # of fixed process spawn however short the sentence.
speechbrain>=1.0           # ECAPA-TDNN speaker identity (config: stt.speaker_gate).
                           # Here rather than in requirements.txt only because it shares
                           # kokoro's torch dependency — it has nothing to do with voices.
